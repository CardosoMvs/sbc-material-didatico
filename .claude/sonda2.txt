<script>
setTimeout(function(){
  var d=document.documentElement,b=document.body,lst=[];
  b.querySelectorAll("*").forEach(function(e){
    var r=e.getBoundingClientRect();
    lst.push([r.right,r.width,(e.id&&('#'+e.id))||'',e.className||'',e.tagName]);
  });
  lst.sort(function(a,c){return c[0]-a[0];});
  var top=lst.slice(0,6).map(function(x){return 'R'+Math.round(x[0])+' w'+Math.round(x[1])+' '+x[4]+' '+x[2]+' '+x[3];}).join(' || ');
  document.title='DBG2 sw='+d.scrollWidth+' zoomOK?'+(b.className||'-')+' '+top;
},600);
</script>
